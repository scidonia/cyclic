\section{Term calculus}
\label{sec:calculus}

Terms are:
\[
t,A ::= x \mid \star_i \mid \Pi x{:}A.\,B \mid \lambda x{:}A.\,t \mid t\,u \mid \mu x{:}A.\,t \mid I\,\vec{a} \mid c\,\vec{a} \mid [t\mid\vec{b}]^I_C
\]
where $x$ is a de~Bruijn index, $\star_i$ a universe, $\mu x{:}A.\,t$ a
fixpoint ($x$ the recursive call), $I$ an inductive type, $c$ a constructor,
and $[t\mid\vec{b}]^I_C$ elimination with motive $C$.
Arrow-annotated letters $\vec{a},\vec{b}$ denote vectors of terms.
$\Sigma$ records inductive signatures.

\paragraph{Contexts and substitutions.}
A context $\Gamma$ is a \emph{dependent telescope}
($\Gamma ::= \cdot \mid \Gamma,A$) where each type may mention earlier
variables; variable $x$ has type $\Gamma(x)=A_x\uparrow^{x+1}$ (shifted for
intervening binders).  A \emph{typed substitution}
$\Delta\vdash\sigma:\Gamma$ is a list $[u_0,\dots,u_{n-1}]$ with
$\Delta\vdash u_i:\sigma(A_i{\uparrow})$ for each $i$, satisfying the
substitution lemma $\Gamma\vdash t:A \Rightarrow \Delta\vdash
t[\sigma]:A[\sigma]$ (mechanised as \texttt{has\_type\_subst}).

\paragraph{Typing.}
$\Sigma;\Gamma\vdash t:A$ is standard for
CIC~\cite{coquand-huet-1988-coc,paulin-mohring-1993-inductive};
$\Sigma$ is implicit below, $\uparrow$ denotes shift.
\[
\begin{array}{c@{\quad}c}
\dfrac{}{\Gamma\vdash x:A}\;{\scriptstyle(\textsc{Var})}\;(\Gamma(x){=}A)
&
\dfrac{}{\Gamma\vdash\star_i:\star_{i+1}}\;{\scriptstyle(\textsc{Sort})}
\\[14pt]
\dfrac{A:\star_i\quad A{::}\Gamma\vdash B:\star_j}%
      {\Gamma\vdash\Pi x{:}A.B:\star_{\max(i,j)}}\;{\scriptstyle(\Pi)}
&
\dfrac{A:\star_i\quad A{::}\Gamma\vdash t:B}%
      {\Gamma\vdash\lambda x{:}A.t:\Pi x{:}A.B}\;{\scriptstyle(\lambda)}
\\[14pt]
\dfrac{t:\Pi x{:}A.B\quad u:A}%
      {\Gamma\vdash t\,u:B[u/x]}\;{\scriptstyle(@)}
&
\dfrac{A:\star_i\quad A{::}\Gamma\vdash t:A\uparrow}%
      {\Gamma\vdash\mu x{:}A.t:A}\;{\scriptstyle(\mu)}
\\[14pt]
\dfrac{\Sigma(I)=\Sigma_I}%
      {\Gamma\vdash I:\star_{\ell(\Sigma_I)+1}}\;{\scriptstyle(\textsc{Ind})}
&
\dfrac{\Sigma(I){=}\Sigma_I\quad \Sigma_I(c){=}\tau_c\quad
       a_i:\tau_{c,i}\;\forall i\quad r_j:I\;\forall j}%
      {\Gamma\vdash c\,\vec a:I}\;{\scriptstyle(\textsc{Roll})}
\\[14pt]
\multicolumn{2}{c}{%
  \dfrac{t:I\,\vec\delta\quad
         x{:}I\,\vec\delta,\Gamma\vdash C:\star_i\quad
         b_c:\Pi\vec\tau_c.\,C[c\,\vec y/x]\;\forall c}%
        {\Gamma\vdash[t\mid\vec b]^I_C:C[t/x]}\;{\scriptstyle(\textsc{Case})}
}
\end{array}
\]

\paragraph{Reduction.}
CBN weak-head: $(\lambda x{:}A.t)\,u\to t[u/x]$,\;
$\mu x{:}A.t\to t[\mu x{:}A.t/x]$,\;
$[c\,\vec a\mid\vec b]^I_C\to b_c\,\vec a$,
plus congruence under application and case scrutinee.
$\to^*$ is $\steps$; $t\Downarrow v$ means $t\steps v$, $v$ a value.

\paragraph{Observational equivalence.}
$t\approx u$ iff for all closing $\sigma$ and values $v$,
$t[\sigma]\Downarrow v\Leftrightarrow u[\sigma]\Downarrow v$
(CIU; Claim~3 criterion).
