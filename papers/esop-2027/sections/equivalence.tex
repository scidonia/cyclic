\section{Main results}
\label{sec:equiv}

We organise the central results as a four-claim pipeline. All results are fully mechanised in Rocq with 0 axioms.

\paragraph{Notation.}
We define the following abbreviations for the key functions, with their Coq counterparts:

\[
\begin{array}{lll}
\mathcal{S}(f,\Sigma,\Gamma,t,A) & \equiv & \texttt{SC.supercompile\_jTy}\;f\;\Sigma\;\Gamma\;t\;A \\
\mathcal{S}_{\!\mathit{tc}}(f,\Sigma,\Gamma,t,A) & \equiv & \texttt{SC.supercompile\_jTy\_tc}\;f\;\Sigma\;\Gamma\;t\;A \\
\mathcal{R}(f_r,\Sigma,\mathit{b},v) & \equiv & \texttt{SC.residualise\_cfg}\;f_r\;\Sigma\;\mathit{b}\;v\;0\;\emptyset \\
t \approx u & \equiv & t \approx_{\mathrm{ciu}} u \quad (\forall\sigma,w.\; t[\sigma] \Downarrow w \iff u[\sigma] \Downarrow w)
\end{array}
\]

\paragraph{Pre-proofs and cyclic proofs.}
A $\mathsf{cfg\_builder}$ $b$ is a finite directed graph where each vertex $v$ carries a judgement $\mathsf{label}(b,v)$ and a successor list $\mathsf{succ}(b,v)$. It is a \emph{rooted pre-proof} if every vertex satisfies a local sequent rule: $\mathsf{label}(b,v)$ is the conclusion of a rule whose premises are $\mathsf{label}(b,\mathsf{succ}(b,v))$. It is a \emph{rooted cyclic proof} if additionally every directed cycle contains a \emph{progress event} (a case-split on a neutral scrutinee), witnessed by a well-founded ranking. We write $\mathcal{P}(b,v)$ and $\mathcal{C}(b,v)$ for these properties.

\subsection{Claim 1: Pre-proof correspondence}

\begin{theorem}[Pre-proof construction]
\label{thm:pre-proof}
For every environment $\Sigma$, fuel $f$, context $\Gamma$, term $t$, and type $A$,
\[
  \mathcal{S}(f,\Sigma,\Gamma,t,A) = \mathsf{Some}\,(v,\mathit{b})
  \;\Longrightarrow\;
  \mathcal{P}(\mathit{b},v).
\]
\end{theorem}

\begin{proof}[Proof sketch]
The supercompiler $\mathsf{SC.supercompile\_cfg}$ is a recursive graph-construction loop. At each step it either marks a memo hit (cyclic backlink), applies driving (asynchronous reduction), splits on a neutral scrutinee (synchronous left rule), or generalises (cut introduction). These four cases correspond exactly to the four rule forms of the focused sequent calculus. The correspondence is proved by three lemmas, each establishing a bisimulation between SC graph edges and sequent rule instances:
\begin{itemize}[leftmargin=*]
\item \texttt{drive\_corresponds\_to\_async\_edge}: a driving edge matches an asynchronous rule;
\item \texttt{split\_corresponds\_to\_sync\_edge}: a case-split edge matches a synchronous rule;
\item \texttt{memo\_corresponds\_to\_fold}: a memo-table hit matches a cyclic backlink.
\end{itemize}
A fourth lemma \texttt{cfg\_vertex\_drive\_rule} classifies each vertex by its successor structure (0, 1 with driving, 1 without change, $n>1$), giving a complete local rule assignment. The overall property follows by structural induction on $f$.
\end{proof}

\subsection{Claim 2: Cyclic proof via budget trace}

\begin{theorem}[Cyclic proof]
\label{thm:cyclic-proof}
For every $\Sigma$, $f$, $\Gamma$, $t$, $A$,
\[
  \mathcal{S}_{\!\mathit{tc}}(f,\Sigma,\Gamma,t,A) = \mathsf{Some}\,(v,\mathit{b})
  \;\Longrightarrow\;
  \mathcal{C}(\mathit{b},v).
\]
\end{theorem}

\begin{proof}[Proof sketch]
The trace condition $\mathsf{SC.trace\_condition\_ok}$ verifies that the non-progress graph (with outgoing edges of progress vertices removed) contains no directed cycles. Each vertex classified as a case-split is a progress vertex. The proof proceeds in two stages:
\begin{enumerate}[leftmargin=*]
\item \textbf{Cycle detection.} We prove \texttt{trace\_condition\_ok\_cycle\_progress}: if $\mathsf{trace\_condition\_ok}$ holds, every directed cycle contains at least one progress edge. The proof uses a combinatorial pigeonhole argument: any walk of length $>|V|$ through $|V|$ vertices repeats, yielding a shorter cycle.
\item \textbf{Budget instrumentation.} We lift the base graph to pairs $(v,k)$ where $k \in [0,|V|]$. Progress edges consume budget ($k \mapsto k-1$); non-progress edges preserve it ($k \mapsto k$). Since every original cycle contains a progress edge, the lifted graph is acyclic; its topological ordering yields a ranking $\lambda(v,k).\;k$, satisfying the well-foundedness condition.
\end{enumerate}
$\sim$700 lines total (FiniteDigraph, SCC, bounded-return lemma, budget instrumentation).
\end{proof}

\subsection{Claim 3: CIU soundness}

\begin{theorem}[CIU equivalence]
\label{thm:ciu}
For every $\Sigma$, $f_s$, $f_r$, $\Gamma$, $t$, $A$,
\[
  \mathcal{S}_{\!\mathit{tc}}(f_s,\Sigma,\Gamma,t,A) = \mathsf{Some}\,(v,\mathit{b})
  \;\Longrightarrow\;
  t \approx \mathcal{R}(f_r,\Sigma,\mathit{b},v).
\]
\end{theorem}

\begin{proof}[Proof sketch]
By fuel induction on the residualiser, with the trace condition guaranteeing exclusion of non-terminating self-loops. The proof decomposes into congruence lemmas for each term constructor:
\begin{itemize}[leftmargin=*]
\item \texttt{ciu\_tApp}: CIU is a congruence for application;
\item \texttt{ciu\_subst0}, \texttt{ciu\_shift}: CIU preserved under substitution and binder shifting;
\item \texttt{ciu\_apps}: CIU preserved under multi-argument application;
\item \texttt{ciu\_case}: pairwise CIU-equivalent branches yield CIU-equivalent case expressions;
\item \texttt{ciu\_of\_eq}: syntactic equality implies CIU.
\end{itemize}
These lift CIU through all nine term constructors. The final step composes them: each residualisation step preserves equivalence up to \texttt{tFix} introduction, and \texttt{tFix} is justified by the cyclic graph structure (the backlink is CIU-equivalent to unfolding by Claim~2). The typed variant follows via the substitution lemma \texttt{ciu\_jTy\_of\_ciu\_jTy\_rel\_eq}.
\end{proof}

\subsection{Claim 4: Example validation}

\begin{proposition}[Examples]
\label{prop:examples}
The following identities pass the pipeline ($\texttt{vm\_compute}$ confirms definitional equality of residuals):
\begin{align*}
  \mathsf{length}\;(\mathsf{map}\;f\;l) &= \mathsf{length}\;l \\
  \mathsf{length}\;(\mathsf{append}\;l_1\;l_2) &= \mathsf{plus}\;(\mathsf{length}\;l_1)\;(\mathsf{length}\;l_2) \\
  \mathsf{length}\;(\mathsf{append}\;(\mathsf{take}\;n\;l)\;(\mathsf{drop}\;n\;l)) &= \mathsf{length}\;l \\
  \mathsf{length}\;(\mathsf{map}\;f\;(\mathsf{map}\;g\;l)) &= \mathsf{length}\;l \\
  \mathsf{Vec}\;A\;(\mathsf{length}\;(\mathsf{map}\;f\;v)) &\equiv \mathsf{Vec}\;A\;(\mathsf{length}\;v)
\end{align*}
Append-associativity additionally holds via 2-pass supercompilation.
\end{proposition}

\begin{proof}
For each example, the supercompiler is invoked with sufficient fuel and the residuals are compared by $\texttt{vm\_compute}$. The CIU guarantee follows from Claim~3.
\end{proof}
