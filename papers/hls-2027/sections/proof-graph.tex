\subsection{Cyclic proof for reverse-append fusion}
\label{sec:reverse-append-proof}

\begin{figure}[t]
\centering\scriptsize
\begin{prooftree}
  \hypo{ys \mapsto \mathsf{nil} \;:\; [ys] \to_s [xs,ys]}
  \infer1[\textsc{back}\;\textcolor{red!70}{\dagger}]{
    \textcolor{red!60}{
      xs,ys \vdash \mathsf{reverse}\;(\mathsf{append}\;xs\;\mathsf{nil}) : \mathsf{List}
      \;\textcolor{red!70}{\dagger}
    }
  }
  \hypo{
    \text{cons branch (below)}
  }
  \infer2[\textsc{iota}]{
    xs,ys \vdash \mathsf{revAcc}\;ys\;(\mathsf{reverse}\;xs) : \mathsf{List}
  }
  \infer1[\textsc{lemma}]{
    xs,ys \vdash \mathsf{revAcc}\;(\mathsf{append}\;xs\;ys)\;\mathsf{nil} : \mathsf{List}
  }
  \infer1[\textsc{fix}]{
    xs,ys \vdash \mathsf{reverse}\;(\mathsf{append}\;xs\;ys) : \mathsf{List}
    \;\textcolor{red!70}{\dagger}
  }
\end{prooftree}

\bigskip

%% Cons branch (nested within Fig 1)
\begin{prooftree}
  \hypo{ys' \mapsto \mathsf{nil} \;:\; [ys'] \to_s [xs,ys]}
  \infer1[\textsc{back}\;\textcolor{red!70}{\dagger}]{
    \textcolor{red!60}{
      xs,y{:}\mathsf{Nat},ys \vdash
      \mathsf{reverse}\;xs : \mathsf{List}
      \;\textcolor{red!70}{\dagger}
    }
  }
  \hypo{\text{(induction on }ys''\text{)}}
  \infer2[\textsc{iota}]{
    xs,y{:}\mathsf{Nat},ys',ys \vdash
    \mathsf{revAcc}\;ys'\;(\mathsf{cons}\;y\;(\mathsf{reverse}\;xs)) : \mathsf{List}
  }
\end{prooftree}

\caption{
  Cyclic proof for $\mathsf{reverse}\;(\mathsf{append}\;xs\;ys)$
  (two proof trees; both backlink to the same root $\dagger$ at bottom).
  \textsc{lemma}: LLM cut
  $\mathsf{revAcc}\;(\mathsf{append}\;xs\;ys)\;acc
  = \mathsf{revAcc}\;ys\;(\mathsf{revAcc}\;xs\;acc)$,
  validated by sub-prover.
  Top tree: case on $ys$. Nil branch closed by \textsc{back} ($ys\mapsto\mathsf{nil}$).
  Bottom tree: cons branch, same structure on $ys'$.
  Its nil branch $\mathsf{reverse}\;xs$ is an instance of the root
  $\mathsf{reverse}\;(\mathsf{append}\;xs\;ys)$ under $ys\mapsto\mathsf{nil}$
  (the same backlink as above).
  The cons branch ($ys''$) repeats the pattern; termination by structural
  induction on the cons chain.
}
\label{fig:reverse-append-proof}
\end{figure}

Without the \textsc{lemma}, the step from
$\mathsf{revAcc}\;(\mathsf{append}\;xs\;ys)\;\mathsf{nil}$
to $\mathsf{revAcc}\;ys\;(\mathsf{reverse}\;xs)$ is blocked ---
the two terms do not anti-unify, and no fold is possible.
\medskip
