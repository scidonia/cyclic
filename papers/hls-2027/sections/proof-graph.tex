\subsection{Cyclic proof for reverse-append fusion}
\label{sec:reverse-append-proof}

\begin{figure}[t]
\centering\scriptsize

%% Main proof
\begin{prooftree}
  \hypo{ys \mapsto \mathsf{nil} \;:\; [ys] \to_s [xs,ys]}
  \infer1[\textsc{back}]{
    \textcolor{red!60}{
      xs,ys \vdash \mathsf{reverse}\;(\mathsf{append}\;xs\;\mathsf{nil}) : \mathsf{List}
      \;\textcolor{red!70}{\dagger}
    }
  }
  \hypo{\textcolor{green!50!black}{
    xs,y{:}\mathsf{Nat},ys',ys \vdash
    \mathsf{revAcc}\;ys'\;(\mathsf{cons}\;y\;(\mathsf{reverse}\;xs)) : \mathsf{List}
  }}
  \infer2[\textsc{iota}\;\textcolor{red!70}{\star}]{
    xs,ys \vdash \mathsf{revAcc}\;ys\;(\mathsf{reverse}\;xs) : \mathsf{List}
  }
  \infer1[\textsc{lemma}]{
    xs,ys \vdash \mathsf{revAcc}\;(\mathsf{append}\;xs\;ys)\;\mathsf{nil} : \mathsf{List}
  }
  \infer1[\textsc{fix}]{
    xs,ys \vdash \mathsf{reverse}\;(\mathsf{append}\;xs\;ys) : \mathsf{List}
    \;\textcolor{red!70}{\dagger}
  }
\end{prooftree}

\medskip

%% Split on ys'
\begin{prooftree}
  \hypo{ys' \mapsto \mathsf{nil} \;:\; [ys'] \to_s [xs,ys]}
  \infer1[\textsc{back}]{
    \textcolor{red!60}{
      xs,y{:}\mathsf{Nat},ys \vdash
      \mathsf{reverse}\;(\mathsf{append}\;xs\;(\mathsf{cons}\;y\;\mathsf{nil})) : \mathsf{List}
      \;\textcolor{red!70}{\dagger}
    }
  }
  \hypo{ys' \mapsto \mathsf{cons}\;y'\;ys'' \;:\; [ys'] \to_s [xs,ys]}
  \infer1[\textsc{back}]{
    \textcolor{green!50!black}{
      xs,y{:}\mathsf{Nat},ys',ys \vdash
      \mathsf{revAcc}\;ys'\;(\mathsf{cons}\;y\;(\mathsf{reverse}\;xs)) : \mathsf{List}
    }
  }
  \infer2[\textsc{iota}\;\textcolor{red!70}{\star}]{
    \textcolor{green!50!black}{
      xs,y{:}\mathsf{Nat},ys',ys \vdash
      \mathsf{revAcc}\;ys'\;(\mathsf{cons}\;y\;(\mathsf{reverse}\;xs)) : \mathsf{List}
    }
  }
\end{prooftree}

\medskip

{\footnotesize
\textbf{Fig.~\ref{fig:reverse-append-proof}.}
Cyclic proof for $\mathsf{reverse}\;(\mathsf{append}\;xs\;ys)$.
$\dagger$ = companion (all backlinks point here).
\textsc{fix}: unfold $\mathsf{reverse}$.
\textsc{lemma}: LLM cut
$\mathsf{revAcc}\;(\mathsf{append}\;xs\;ys)\;acc
= \mathsf{revAcc}\;ys\;(\mathsf{revAcc}\;xs\;acc)$,
validated by sub-prover.
\textsc{iota}: case on $ys$ (top) or $ys'$ (bottom).
\textsc{back}: the nil branch (at any depth) is an instance of the root
companion under a suitable substitution:
$ys\mapsto\mathsf{nil}$ (top), $ys\mapsto\mathsf{cons}\;y\;\mathsf{nil}$ (bottom).
The cons branch repeats the pattern; termination by structural induction on
the cons chain (every split removes one $\mathsf{cons}$).
}

\caption{See caption text above.}
\label{fig:reverse-append-proof}
\end{figure}

Without the \textsc{lemma}, the step from
$\mathsf{revAcc}\;(\mathsf{append}\;xs\;ys)\;\mathsf{nil}$
to $\mathsf{revAcc}\;ys\;(\mathsf{reverse}\;xs)$ is blocked ---
the two terms do not anti-unify, and no fold is possible.
