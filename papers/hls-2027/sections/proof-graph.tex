\subsection{Cyclic proof for reverse-append fusion}
\label{sec:reverse-append-proof}

\begin{figure}[t]
\centering\scriptsize
\begin{prooftree}
  \hypo{xs,ys \vdash \mathsf{reverse}\;xs : \mathsf{List}}
  \hypo{ys \mapsto \mathsf{nil} \;:\; [ys] \to_s [xs,ys]}
  \infer2[\textsc{back}\;\textcolor{red!70}{\dagger}]{
    \textcolor{red!60}{
      xs,ys \vdash \mathsf{reverse}\;(\mathsf{append}\;xs\;\mathsf{nil}) : \mathsf{List}
      \;\textcolor{red!70}{\dagger}
    }
  }
  \hypo{\textcolor{green!60!black}{
    xs,y{:}\mathsf{Nat},ys'
    \vdash \mathsf{revAcc}\;ys'\;(\mathsf{cons}\;y\;(\mathsf{reverse}\;xs)) : \mathsf{List}
    \;\textit{(see Fig.~\ref{fig:drive-proof})}
  }}
  \infer2[\textcolor{red!70}{\textsc{split}\;$\star$}]{
    xs,ys \vdash \mathsf{revAcc}\;ys\;(\mathsf{reverse}\;xs) : \mathsf{List}
  }
\end{prooftree}
\caption{Part~1: split on $ys$.
$\dagger$\,=\,backlink; red\,=\,companion; $\star$\,=\,progress.
Nil branch closes via \textsc{back} with substitution $ys\mapsto\mathsf{nil}$.
Cons branch (green) continues in Fig.~\ref{fig:drive-proof}.}
\label{fig:split-proof}
\end{figure}

\begin{figure}[t]
\centering\scriptsize
\begin{prooftree}
  \hypo{
    \textcolor{green!60!black}{\cdots\;\text{(from Fig.~\ref{fig:split-proof} split)}}
  }
  \infer1[\phantom{X}]{
    xs,ys \vdash \mathsf{revAcc}\;ys\;(\mathsf{reverse}\;xs) : \mathsf{List}
  }
  \infer1[\textsc{lemma}]{
    xs,ys \vdash \mathsf{revAcc}\;(\mathsf{append}\;xs\;ys)\;\mathsf{nil} : \mathsf{List}
  }
  \infer1[\textsc{drv}]{
    xs,ys \vdash \mathsf{reverse}\;(\mathsf{append}\;xs\;ys) : \mathsf{List}
    \;\textcolor{red!70}{\dagger}
  }
\end{prooftree}
\caption{Part~2: driving from root to split (reading bottom-up).
\textsc{drv}: unfold $\mathsf{reverse} = \lambda l.\,\mathsf{revAcc}\;l\;\mathsf{nil}$.
\textsc{lemma}: LLM cut
$\mathsf{revAcc}\;(\mathsf{append}\;xs\;ys)\;acc
= \mathsf{revAcc}\;ys\;(\mathsf{revAcc}\;xs\;acc)$,
validated by sub-prover.
Green $\cdots$ connects back to the conclusion of Fig.~\ref{fig:split-proof}.}
\label{fig:drive-proof}
\end{figure}

Without the \textsc{lemma}, the step from
$\mathsf{revAcc}\;(\mathsf{append}\;xs\;ys)\;\mathsf{nil}$
to $\mathsf{revAcc}\;ys\;(\mathsf{reverse}\;xs)$ is blocked ---
the two terms do not anti-unify, and no fold is possible.
