\subsection{Cyclic proof for reverse-append fusion}
\label{sec:reverse-append-proof}

\begin{figure}[t]
\centering
\scriptsize
\[
\AxiomC{$
  xs, ys \vdash
  \mathsf{reverse}\;(\mathsf{append}\;xs\;ys) : \mathsf{List}
  \;\textcolor{red!70}{\dagger}
$}
\RightLabel{\;\textsc{fix}$+\beta$}
\UnaryInfC{$
  xs, ys \vdash
  \mathsf{revAcc}\;(\mathsf{append}\;xs\;ys)\;\mathsf{nil} : \mathsf{List}
$}
\RightLabel{\;\textsc{lemma}}
\UnaryInfC{$
  xs, ys \vdash
  \mathsf{revAcc}\;ys\;(\mathsf{reverse}\;xs) : \mathsf{List}
$}
\DisplayProof
\]
\caption{Part~1: driving phase. \textsc{lemma} = LLM cut
$\mathsf{revAcc}\;(\mathsf{append}\;xs\;ys)\;acc
= \mathsf{revAcc}\;ys\;(\mathsf{revAcc}\;xs\;acc)$.
See Fig.~\ref{fig:split-proof} for the split.}
\label{fig:drive-proof}
\end{figure}

\begin{figure}[t]
\centering
\scriptsize
\[
\AxiomC{$xs,ys \vdash \mathsf{reverse}\;xs : \mathsf{List}$}
\RightLabel{\;\textsc{back}\;\textcolor{red!70}{\dagger}}
\UnaryInfC{\textcolor{red!60}{$
  xs,ys \vdash \mathsf{reverse}\;(\mathsf{append}\;xs\;\mathsf{nil}) : \mathsf{List}
  \;\textcolor{red!70}{\dagger}
$}}
\AxiomC{\textcolor{green!60!black}{$\vdots$}}
\RightLabel{\;\textcolor{red!70}{\textsc{split}\;$\star$}}
\BinaryInfC{$
  xs, ys \vdash
  \mathsf{revAcc}\;ys\;(\mathsf{reverse}\;xs) : \mathsf{List}
$}
\DisplayProof
\]
\caption{Part~2: split on $ys$ (connects to Fig.~\ref{fig:drive-proof}).
$\dagger$\,=\,backlink; red\,=\,companion; $\star$\,=\,progress; green $\vdots$\,=\,cons
branch proved by structural induction on $ys'$ (same pattern).
Nil closes via \textsc{back}: $\mathsf{reverse}\;xs
= \mathsf{reverse}\;(\mathsf{append}\;xs\;\mathsf{nil})$
under $ys{\mapsto}\mathsf{nil}$.}
\label{fig:split-proof}
\end{figure}

Without the \textsc{lemma}, the step from
$\mathsf{revAcc}\;(\mathsf{append}\;xs\;ys)\;\mathsf{nil}$
to $\mathsf{revAcc}\;ys\;(\mathsf{reverse}\;xs)$ is blocked ---
the two terms do not anti-unify, and no fold is possible.
