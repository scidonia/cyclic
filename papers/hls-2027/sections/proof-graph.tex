\subsection{Cyclic proof for reverse-append fusion}
\label{sec:reverse-append-proof}

Figure~\ref{fig:reverse-append-proof} shows the cyclic proof.
The root goal (bottom) is $\mathsf{reverse}\;(\mathsf{append}\;xs\;ys) : \mathsf{List}$.
\textsc{fix} unfolds $\mathsf{reverse}$; the LLM-proposed \textsc{lemma} rewrites
$\mathsf{revAcc}\;(\mathsf{append}\;xs\;ys)\;\mathsf{nil}$ to
$\mathsf{revAcc}\;ys\;(\mathsf{reverse}\;xs)$; \textsc{iota} case-reduces on
$ys$.  The nil branch closes via \textsc{back} with substitution
$ys\mapsto\mathsf{nil}$.  The cons branch (green) is proved by the same
cyclic structure on $ys'$ (Figure~\ref{fig:cons-branch}).

Without the \textsc{lemma}, the step from
$\mathsf{revAcc}\;(\mathsf{append}\;xs\;ys)\;\mathsf{nil}$
to $\mathsf{revAcc}\;ys\;(\mathsf{reverse}\;xs)$ is blocked ---
the two terms do not anti-unify, and no fold is possible.

\begin{figure}[t]
\centering\scriptsize
\begin{prooftree}
  \hypo{ys \mapsto \mathsf{nil} \;:\; [ys] \to_s [xs,ys]}
  \infer1[\textsc{back}\;\textcolor{red!70}{\dagger}]{
    \textcolor{red!60}{
      xs,ys \vdash \mathsf{reverse}\;(\mathsf{append}\;xs\;\mathsf{nil}) : \mathsf{List}
      \;\textcolor{red!70}{\dagger}
    }
  }
  \hypo{
    \textcolor{green!60!black}{
      xs,y{:}\mathsf{Nat},ys' \vdash
      \mathsf{revAcc}\;ys'\;(\mathsf{cons}\;y\;(\mathsf{reverse}\;xs)) : \mathsf{List}
      \quad\text{(Fig.~\ref{fig:cons-branch})}
    }
  }
  \infer2[\textsc{iota}]{
    xs,ys \vdash \mathsf{revAcc}\;ys\;(\mathsf{reverse}\;xs) : \mathsf{List}
  }
  \infer1[\textsc{lemma}]{
    xs,ys \vdash \mathsf{revAcc}\;(\mathsf{append}\;xs\;ys)\;\mathsf{nil} : \mathsf{List}
  }
  \infer1[\textsc{fix}]{
    xs,ys \vdash \mathsf{reverse}\;(\mathsf{append}\;xs\;ys) : \mathsf{List}
    \;\textcolor{red!70}{\dagger}
  }
\end{prooftree}
\caption{
  Cyclic proof for $\mathsf{reverse}\;(\mathsf{append}\;xs\;ys)$.
  $\dagger$ = backlink; red = companion.
  \textsc{lemma}: LLM cut (validated by sub-prover).
  Nil branch: \textsc{back} under $ys\mapsto\mathsf{nil}$.
  Cons branch: Fig.~\ref{fig:cons-branch}.
}
\label{fig:reverse-append-proof}
\end{figure}

\begin{figure}[t]
\centering\scriptsize
\begin{prooftree}
  \hypo{ys' \mapsto \mathsf{nil} \;:\; [ys'] \to_s [xs,ys,y]}
  \infer1[\textsc{back}\;\textcolor{red!70}{\dagger}]{
    \textcolor{red!60}{
      xs,y{:}\mathsf{Nat},ys \vdash
      \mathsf{reverse}\;(\mathsf{append}\;xs\;(\mathsf{cons}\;y\;\mathsf{nil}))
      : \mathsf{List}
      \;\textcolor{red!70}{\dagger}
    }
  }
  \hypo{
    \text{cons branch: induction on }ys''\text{ (same structure)}
  }
  \infer2[\textsc{iota}]{
    xs,y{:}\mathsf{Nat},ys',ys \vdash
    \mathsf{revAcc}\;ys'\;(\mathsf{cons}\;y\;(\mathsf{reverse}\;xs)) : \mathsf{List}
  }
  \infer1[\textsc{lemma}]{
    xs,y{:}\mathsf{Nat},ys',ys \vdash
    \mathsf{revAcc}\;(\mathsf{append}\;xs\;(\mathsf{cons}\;y\;\mathsf{nil}))\;\mathsf{nil}
    : \mathsf{List}
  }
  \infer1[\textsc{fix}]{
    xs,y{:}\mathsf{Nat},ys \vdash
    \mathsf{reverse}\;(\mathsf{append}\;xs\;(\mathsf{cons}\;y\;\mathsf{nil}))
    : \mathsf{List}
    \;\textcolor{red!70}{\dagger}
  }
\end{prooftree}
\caption{
  Cons branch proof (isomorphic to Fig.~\ref{fig:reverse-append-proof}
  with $ys \leftarrow \mathsf{cons}\;y\;\mathsf{nil}$).
  The root $\mathsf{reverse}\;(\mathsf{append}\;xs\;(\mathsf{cons}\;y\;\mathsf{nil}))\dagger$
  is the same root as Fig.~\ref{fig:reverse-append-proof} under the substitution
  $ys \mapsto \mathsf{cons}\;y\;\mathsf{nil}$.
  The nil branch closes via \textsc{back} under $ys'\mapsto\mathsf{nil}$.
  The cons branch recurses on $ys''$ — same proof structure.
}
\label{fig:cons-branch}
\end{figure}
